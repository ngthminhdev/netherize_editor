pub fn two_sum(nums: Vec<i32>, target: i32) -> Vec<i32> {
let mut seen = std::collections::HashMap::new();
for (i, &v) in nums.iter().enumerate() {
let c = target - v; 
if let Some(&j) = seen.get(&c) { return vec![j as i32, i as i32] }
seen.insert(v, i);
}
vec![]
}
